func TestReviewFailedUploadPreservesDirectory(t *testing.T) {
	scope := t.TempDir()
	victim := filepath.Join(scope, "directory", "keep.txt")
	if err := os.MkdirAll(filepath.Dir(victim), 0755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(victim, []byte("original"), 0600); err != nil {
		t.Fatal(err)
	}
	key := []byte("review-key")
	perm := users.Permissions{Create: true, Modify: true}
	st := scopedUserStorage(t, scope, perm, key)
	req := httptest.NewRequest(http.MethodPost, "/directory?override=true", strings.NewReader("data"))
	req.Header.Set("X-Auth", signToken(t, perm, key))
	rec := httptest.NewRecorder()
	handle(resourcePostHandler(diskcache.NewNoOp()), "", st, &settings.Server{}).ServeHTTP(rec, req)
	if _, err := os.Stat(victim); err != nil {
		t.Fatalf("upload returned %d and deleted existing directory contents despite Delete=false: %v", rec.Code, err)
	}
}

func TestReviewInvalidTusLengthPreservesExistingFile(t *testing.T) {
	scope := t.TempDir()
	target := filepath.Join(scope, "keep.txt")
	if err := os.WriteFile(target, []byte("original"), 0600); err != nil {
		t.Fatal(err)
	}
	key := []byte("review-key")
	perm := users.Permissions{Create: true, Modify: true}
	st := scopedUserStorage(t, scope, perm, key)
	cache := newMemoryUploadCache()
	t.Cleanup(cache.Close)
	req := httptest.NewRequest(http.MethodPost, "/keep.txt?override=true", http.NoBody)
	req.Header.Set("X-Auth", signToken(t, perm, key))
	req.Header.Set("Upload-Length", "invalid")
	rec := httptest.NewRecorder()
	handle(tusPostHandler(cache), "", st, &settings.Server{}).ServeHTTP(rec, req)
	if rec.Code != http.StatusBadRequest {
		t.Fatalf("got %d, want 400", rec.Code)
	}
	content, err := os.ReadFile(target)
	if err != nil {
		t.Fatal(err)
	}
	if string(content) != "original" {
		t.Fatalf("rejected TUS POST altered existing file: got %q", content)
	}
}

type reviewGatedBody struct {
	io.Reader
	once    sync.Once
	arrived chan<- struct{}
	release <-chan struct{}
}

func (r *reviewGatedBody) Read(p []byte) (int, error) {
	r.once.Do(func() { r.arrived <- struct{}{}; <-r.release })
	return r.Reader.Read(p)
}

func TestReviewConcurrentTusEnforcesLength(t *testing.T) {
	scope := t.TempDir()
	key := []byte("review-key")
	perm := users.Permissions{Create: true, Modify: true}
	st := scopedUserStorage(t, scope, perm, key)
	token := signToken(t, perm, key)
	cache := newMemoryUploadCache()
	t.Cleanup(cache.Close)
	post := httptest.NewRequest(http.MethodPost, "/file.txt", http.NoBody)
	post.Header.Set("X-Auth", token)
	post.Header.Set("Upload-Length", "5")
	rec := httptest.NewRecorder()
	handle(tusPostHandler(cache), "", st, &settings.Server{}).ServeHTTP(rec, post)
	if rec.Code != http.StatusCreated {
		t.Fatalf("POST got %d", rec.Code)
	}
	location, err := url.Parse(rec.Header().Get("Location"))
	if err != nil {
		t.Fatal(err)
	}
	uploadTarget := "/file.txt"
	if location.RawQuery != "" {
		uploadTarget += "?" + location.RawQuery
	}
	arrived := make(chan struct{}, 2)
	release := make(chan struct{})
	done := make(chan int, 2)
	handler := handle(tusPatchHandler(cache), "", st, &settings.Server{})
	for _, body := range []string{"hello", "world"} {
		go func(body string) {
			req := httptest.NewRequest(http.MethodPatch, uploadTarget, &reviewGatedBody{Reader: bytes.NewBufferString(body), arrived: arrived, release: release})
			req.Header.Set("X-Auth", token)
			req.Header.Set("Content-Type", "application/offset+octet-stream")
			req.Header.Set("Upload-Offset", "0")
			response := httptest.NewRecorder()
			handler.ServeHTTP(response, req)
			done <- response.Code
		}(body)
	}
	select {
	case <-arrived:
	case <-time.After(5 * time.Second):
		close(release)
		t.Fatal("first PATCH did not reach body reader")
	}
	// Give a competing request time to pass the size check while the first
	// body is blocked. A serialized implementation may keep it waiting here.
	select {
	case <-arrived:
	case <-time.After(time.Second):
	}
	close(release)
	first, second := <-done, <-done
	info, err := os.Stat(filepath.Join(scope, "file.txt"))
	if err != nil {
		t.Fatal(err)
	}
	if info.Size() > 5 {
		t.Fatalf("PATCH statuses %d/%d; declared 5 bytes but wrote %d", first, second, info.Size())
	}
}

func TestReviewSymlinkCannotBypassDenyRule(t *testing.T) {
	scope := t.TempDir()
	if err := os.WriteFile(filepath.Join(scope, "secret.txt"), []byte("secret"), 0600); err != nil {
		t.Fatal(err)
	}
	if err := os.Symlink(filepath.Join(scope, "secret.txt"), filepath.Join(scope, "alias.txt")); err != nil {
		t.Skip(err)
	}
	key := []byte("review-key")
	perm := users.Permissions{Download: true}
	st := denyRuleStorage(t, scope, "/secret.txt", perm, key)
	handler := handle(rawHandler, "", st, &settings.Server{})
	for _, target := range []string{"/secret.txt", "/alias.txt"} {
		req := httptest.NewRequest(http.MethodGet, target, http.NoBody)
		req.Header.Set("X-Auth", signToken(t, perm, key))
		rec := httptest.NewRecorder()
		handler.ServeHTTP(rec, req)
		if rec.Code != http.StatusForbidden {
			t.Errorf("GET %s returned %d and body %q; want 403", target, rec.Code, rec.Body.String())
		}
	}
}

func TestReviewCSPProtectsAppHTML(t *testing.T) {
	scope := t.TempDir()
	st := scopedUserStorage(t, scope, users.Permissions{}, []byte("review-key"))
	if err := st.Settings.Save(&settings.Settings{Key: []byte("review-key"), AuthMethod: auth.MethodJSONAuth}); err != nil {
		t.Fatal(err)
	}
	if err := st.Auth.Save(&auth.JSONAuth{}); err != nil {
		t.Fatal(err)
	}
	cache := newMemoryUploadCache()
	t.Cleanup(cache.Close)
	assets := fstest.MapFS{"public/index.html": &fstest.MapFile{Data: []byte("<html>app</html>")}}
	handler, err := NewHandler(nil, diskcache.NewNoOp(), cache, st, &settings.Server{Root: scope}, assets)
	if err != nil {
		t.Fatal(err)
	}
	for _, target := range []string{"/health", "/", "/files/"} {
		rec := httptest.NewRecorder()
		handler.ServeHTTP(rec, httptest.NewRequest(http.MethodGet, target, http.NoBody))
		if rec.Code != http.StatusOK {
			t.Fatalf("GET %s returned %d: %s", target, rec.Code, rec.Body.String())
		}
		if rec.Header().Get("Content-Security-Policy") == "" {
			t.Errorf("GET %s serves application HTML without configured Content-Security-Policy", target)
		}
	}
}
